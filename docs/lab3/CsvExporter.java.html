  1. package drinkshop.export;

  13. public class CsvExporter {

  15.     private CsvExporter() {}
  16.     public static void exportOrders(List<Product> products, List<Order> orders, String path) {
  17.         String date = LocalDate.now().format(DateTimeFormatter.ofPattern("dd.MM.yyyy"));
  18.         // Write main orders CSV (valid tabular format)
  19.         try (FileWriter w = new FileWriter(path)) {
  20.             w.write("OrderId,Product,Quantity,Price\n");
  21.             for (Order o : orders) {
  22.                 for (OrderItem i : o.getItems()) {
  23.                     Product p = products.stream()
  24.                             .filter(prod -> i.getProduct().getId() == prod.getId())
  25.                             .findFirst()
  26.                             .orElse(null);
  27.                     if (p == null) continue;
  28.                     w.write(o.getId() + "," + p.getNume() + "," + i.getQuantity() + "," + i.getTotal() + "\n");
  29.                 }
  30.             }
  31.         } catch (IOException e) {
  32.             throw new RuntimeException(e);
  33.         }

  35.         // Write daily summary to a separate file
  36.         String summaryPath = path.replaceAll("\\.csv$", "") + "_summary_" + date.replace(".", "-") + ".csv";
  37.         try (FileWriter sw = new FileWriter(summaryPath)) {
  38.             sw.write("OrderId,OrderTotal\n");
  39.             double sum = 0.0;
  40.             for (Order o : orders) {
  41.                 sw.write(o.getId() + "," + o.getTotalPrice() + "\n");
  42.                 sum += o.getTotalPrice();
  43.             }
  44.             sw.write("TOTAL," + sum + "\n");
  45.         } catch (IOException e) {
  46.             throw new RuntimeException(e);
  47.         }
  48.     }
  49. }
